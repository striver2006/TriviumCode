## 2. 判据（单点，决策一、四）

### 2.1 导出判定函数（`gate.ts`，放在 `READONLY_BREACH_TRIAGE` 旁）

```ts
/** 只读失守成立之后，按跑后复算分三档（用户决策 2026-10-06，原始需求 20261005 第 1 条） */
export type ReadonlyBreachVerdict = 'confirmed' | 'unconfirmed_unchanged' | 'unconfirmed_unknown';

export function judgeReadonlyBreach(
  drift: Pick<SnapshotMeta['drift'], 'changed' | 'unknown_reason'>,
): ReadonlyBreachVerdict {
  if (drift.unknown_reason !== undefined) return 'unconfirmed_unknown';
  return drift.changed ? 'confirmed' : 'unconfirmed_unchanged';
}
```

- 只在失守成立时调用；「失守是否成立」仍只看 `guard.readonly_breach`（决策五）。
- 先判 `unknown_reason`：写侧只在 `changed: true` 时写它（§1.2），两种次序拦截集相同；先判它是为了「`changed: false` 却带 `unknown_reason`」（手改留痕才可达）时按「比不了」说，不说成「复算无变化」——说法更弱、方向保守。
